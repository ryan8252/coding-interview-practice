## #785 Is Graph Bipartite? — B

使用者 Google「如何判斷Graph 是不是 Bipartite」，看到「使用 BFS 或 DFS 進行逐層或遞迴著色檢查」，之後自行 C++ 實作。核心方向受到外部查詢提示，不算無提示 Medium。Disconnected graph 處理、queue、color map、衝突檢測均自行寫成；程式 AC。外層每連通分量重掃找未染色節點，最壞 Time O(V²+E)，Space O(V)。`n==0` 回傳 false 在 LeetCode 本題非空輸入限制下不影響正確性。

```cpp
class Solution {
public:
    bool isBipartite(vector<vector<int>>& graph) {
        int n=graph.size();
        if(n==0) return false;

        queue<int> q;
        unordered_map<int,int> color;

        while(color.size()<n){
            for(int i=0;i<n;i++){
                if(color.find(i)==color.end()){
                    q.push(i);
                    color[i]=1;
                    break;
                }
            }
            while(!q.empty()){
                int node=q.front();
                q.pop();
                int c=1;
                if(color[node]==c) c=2;
                for(auto &i:graph[node]){
                    if(color.find(i)==color.end()){
                        color[i]=c;
                        q.push(i);
                    }
                    else{
                        if(color[i]==color[node]) return false;
                    }
                }
            }
        }
        return true;
    }
};
```

```cpp
class Solution {
public:
    int change(int amount, vector<int>& coins) {
        vector<int> dp(amount+1,0);
        dp[0]=1;

        for(int i=1;i<amount+1;i++){
            unordered_map<int,int> used;
            for(auto &c:coins){
                if((i-c)==c){
                    dp[i]++;
                    used[c]=1;
                }
                else if((i-c)>=0 && used.find(c)==used.end()){
                    dp[i]+=dp[i-c];
                    used[c]=1;
                    used[i-c]=1;
                }
            }
        }

        for(int i=0;i<amount+1;i++){
            cout<<dp[i]<<" ";
        }
        return dp[amount];
    }
};
```

## #387 First Unique Character in a String — A

03:11 AC，C++ frequency map + first unique index，Time O(n)、Space O(1) fixed lowercase alphabet。程式正確；未回報外部演算法協助。

```cpp
class Solution {
public:
    int firstUniqChar(string s) {
        unordered_map<char,int> m;
        for(auto &c:s){
            m[c]++;
        }
        for(int i=0;i<s.length();i++){
            if(m[s[i]]==1) return i;
        }
        return -1;
    }
};
```
