## #78 Subsets

完全獨立完成。C++ backtracking 正確；時間 O(n·2^n)，額外空間 O(n)（不含輸出）。

```cpp
class Solution {
public:
    void backtrack(int i, vector<int>& nums,vector<vector<int>>& res,vector<int>& sub) {
        res.push_back(sub);
        for(int j=i;j<size(nums);j++){
            sub.push_back(nums[j]);
            backtrack(j+1,nums,res,sub);
            sub.pop_back();
        }
    }
    vector<vector<int>> subsets(vector<int>& nums) {
        vector<vector<int>> res;
        vector<int> sub;
        backtrack(0,nums,res,sub);
        return res;
    }
};
```

## #2264 Largest 3-Same-Digit Number in String

06:54 AC；O(n) time，O(1) space。未明確回報是否使用提示，暫 A。

class Solution {
public:
    string largestGoodInteger(string num) {
        string res="";
        for(int i=2;i<num.length();i++){
            if(num[i-2]==num[i-1] && num[i-1]==num[i]){
                if (res == "" || res[0] < num[i]) {
                    res = string(3, num[i]);
                }
            }
        }
        return res;
    }
};
