## #684 Redundant Connection

第一版：16:43，使用者回報 AC；此版本以兩端點是否都曾出現判斷成環，不保證在同一連通分量。對 edges = [[1,2],[3,4],[2,3],[1,4]] 會在 [2,3] 誤判；且 `edges[-1]` 在 C++ 越界，屬錯誤程式。

```cpp
class Solution {
public:
    vector<int> findRedundantConnection(vector<vector<int>>& edges) {
        unordered_set<int> visited;

        for(auto& edge:edges){
            if( visited.find(edge[0])!=visited.end() && visited.find(edge[1])!=visited.end() ){
                return edge;
            }
            visited.insert(edge[0]);
            visited.insert(edge[1]);
        }

        return edges[-1];
    }
};
```

第二版：22:37，向 GPT 詢問 Union-Find 寫法後提供。已理解原理，但此版本 `parent.resize(n+1)` 後 `for(int i=0;i<n;i++)` 漏初始化 index n；該位置預設 0，可能造成錯誤判斷。須核對 LeetCode 實際提交程式。演算法選擇在第一版有誤，第二版為 implementation bug；評 C，列高優先複習。

```cpp
class Solution {
public:
    vector<int> parent;

    int find(int x){
        if(parent[x] != x){
            parent[x] = find(parent[x]);
        }
        return parent[x];
    }

    bool unite(int x, int y){
        int px=find(x);
        int py=find(y);

        if(px==py) return false;

        parent[px]=py;
        return true;
    }

    vector<int> findRedundantConnection(vector<vector<int>>& edges) {
        int n=edges.size();
        parent.resize(n+1);
        for(int i=0;i<n;i++){
            parent[i]=i;
        }

        for(auto& edge: edges){
            int u=edge[0];
            int v=edge[1];

            if(!unite(u,v)){
                return edge;
            }
        }

        return {};
    }
};
```

## #39 Combination Sum

17:46 AC，完全獨立完成。Backtracking 正確，允許重複選取同一元素且用 start 避免順序重複；push/pop 配對正確。評 A。

```cpp
class Solution {
public:
    void backtracing(vector<int>& candidates, int target, vector<int>& comb, vector<vector<int>>& res, int start){
        for(int i=start;i<candidates.size();i++){
            int next_target =target-candidates[i];
            if(next_target>0){
                comb.push_back(candidates[i]);
                backtracing(candidates, next_target, comb, res, i);
                comb.pop_back();
            }
            else if(next_target==0){
                comb.push_back(candidates[i]);
                res.push_back(comb);
                comb.pop_back();
            }
        }
    }

    vector<vector<int>> combinationSum(vector<int>& candidates, int target) {
        vector<vector<int>> res;
        vector<int> comb;
        backtracing(candidates, target, comb, res, 0);
        return res;
    }
};
```

## #268 Missing Number

04:22 AC，O(n) 時間、O(1) 額外空間，評 A（提示獨立性未特別回報）。整數溢位在本題限制下不是問題。

```cpp
class Solution {
public:
    int missingNumber(vector<int>& nums) {
        int Size=nums.size();
        int ans=Size*(Size+1)/2;

        for(auto &n:nums){
            ans-=n;
        }
        return ans;
    }
};
```
