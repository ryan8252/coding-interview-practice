```cpp
class Solution {
public:
    int shortestPathBinaryMatrix(vector<vector<int>>& grid) {
        int n=grid.size();

        if(grid[0][0]==1 || grid[n-1][n-1]==1) return -1;
        const int dx[8]={1,-1,0,0,1,1,-1,-1};
        const int dy[8]={0,0,1,-1,1,-1,1,-1};

        queue<pair<int,int>> q;
        q.push({0,0});
        int cnt=0;
        int arrived=0;
        while(!q.empty()){
            int qsize=q.size();
            for(int j=0;j<qsize;j++){
                auto node=q.front();
                q.pop();
                grid[node.first][node.second]=1;
                // cout<<node.first<<" "<<node.second<<endl;
                if(node.first==n-1 && node.second==n-1){
                    while(!q.empty()) q.pop();
                    arrived=1;
                    break;
                }

                for(int i=0;i<8;i++){
                    int x=node.first+dx[i];
                    int y=node.second+dy[i];

                    if(x>=0 && x<n && y>=0 && y<n){
                        if(grid[x][y]==0){

                            q.push({x,y});
                        }
                    }
                }
            }
            cnt++;
        }
        if(arrived) return cnt;
        else return -1;
    }
};
```

```cpp
class Solution {
public:
    int shortestPathBinaryMatrix(vector<vector<int>>& grid) {
        int n=grid.size();

        if(grid[0][0]==1 || grid[n-1][n-1]==1) return -1;
        const int dx[8]={1,-1,0,0,1,1,-1,-1};
        const int dy[8]={0,0,1,-1,1,-1,1,-1};
        set<pair<int,int>> visited;
        queue<pair<int,int>> q;
        q.push({0,0});
        visited.insert({0,0});
        int cnt=0;
        int arrived=0;
        while(!q.empty()){
            int qsize=q.size();
            for(int j=0;j<qsize;j++){
                auto node=q.front();
                q.pop();
                grid[node.first][node.second]=1;
                // cout<<node.first<<" "<<node.second<<endl;
                if(node.first==n-1 && node.second==n-1){
                    while(!q.empty()) q.pop();
                    arrived=1;
                    break;
                }

                for(int i=0;i<8;i++){
                    int x=node.first+dx[i];
                    int y=node.second+dy[i];

                    if(x>=0 && x<n && y>=0 && y<n && visited.find({x,y})==visited.end()){
                        if(grid[x][y]==0){
                            q.push({x,y});
                            visited.insert({x,y});
                        }
                    }
                }
            }
            cnt++;
        }
        if(arrived) return cnt;
        else return -1;
    }
};
```

```cpp
class Solution {
public:
    int numSquares(int n) {
        vector<int> dp(n+1,0);
        int s=1;
        dp[1]=1;
        for(int i=2;i<n+1;i++){
            if ((s+1)*(s+1)==i){
                dp[i]=1;
                s++;
            }
            else{
                dp[i]=dp[i-s*s]+1;
            }
        }
        return dp[n];
    }
};
```

### #279 24:06 AC 程式
```cpp
class Solution {
public:
    int numSquares(int n) {
        vector<int> dp(n+1,0);
        
        for (int i = 1; i <= n; i++) {
            dp[i] = n;  // 先設定成一個足夠大的值

            for (int j = 1; j*j <= i; j++) {
                dp[i]=min(dp[i],dp[i-j*j]+1);
            }
        }

        return dp[n];
    }
};
```
