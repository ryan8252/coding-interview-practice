class Solution {
private:
    unordered_map<int,vector<int>> adj;
    set<int> s;
    unordered_map<int,int> visited;
    int label;
public:
    void DFS(int node,int label){
        visited[node]=label;
        label--;
        for(auto &i:adj[node]){
            if(visited.find(i)==visited.end()){
                DFS(i,label);
            }
            else{
                visited[node]=-1;
            }
        }
    }

    vector<int> findOrder(int numCourses, vector<vector<int>>& prerequisites) {
        for(auto &prerequisite:prerequisites){
            adj[prerequisite[1]].push_back(prerequisite[0]);
            s.insert(prerequisite[0]);
        }
        label=numCourses;
        for(int i=0;i<n;i++){
            if(s.find(i)==s.end()){
                DFS(i,label);
            }
        }
    }
};

### 評估與問題分類

- Concept：理解先修 -> 後修的 directed adjacency，嘗試 DFS。
- Algorithm Design：label 遞減不構成合法 topological ordering；visited 的「有無拜訪」不能分辨仍在 DFS recursion path 上的節點與已完成節點；遇到已訪節點直接 -1 會誤判；僅從未出現在先修目標 s 的課程開始，不足以處理全為 cycle 的圖。
