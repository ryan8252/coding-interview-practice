## #740 Delete and Earn — 35:56 AC，C

起初嘗試 Greedy 不成，Gemini 告知「這題可以轉成 House Robber」，此為核心 algorithm reduction 提示；使用者之後自己完成 C++ DP，編碼與狀態轉移正確。

```cpp
class Solution {
public:
    int deleteAndEarn(vector<int>& nums) {
        unordered_map<int,int> m;
        int n=0;
        for(auto &i:nums){
            m[i]++;
            if(i>n) n=i;
        }
        vector<int> dp(n+1,0);
        if(m.find(1)!=m.end()) dp[1]=1*m[1];

        for(int i=2;i<n+1;i++){
            if(m.find(i)!=m.end()){
                dp[i]=max(dp[i-2]+i*m[i],dp[i-1]);
            }
            else dp[i]=dp[i-1];
        }

        return dp[n];
    }
};
```

- Implementation：獲方向後自行建立 frequency map 與 dp，實作正確，無明顯 bug。
- Complexity：令 N=nums.size()，M=max(nums)，平均 O(N+M) time，O(M) space。
- 獨立性：C；即使 AC 也不提升為 A，後續以無參考 reduction 複習。

```cpp
class Solution {
private:
    unordered_map<string, vector<pair<string,double>>> graph;
    unordered_set<string> alphabet;

public:
    double DFS(string node,string target,double product,unordered_set<string>& visited){
        if(node==target) return product;

        visited.insert(node);
        for(auto& p:graph[node]){
            if(visited.find(p.first)==visited.end()){
                double ans=DFS(p.first,target,product*p.second, visited);
                if(ans!=-1.0) return ans;
            }
        }

        return -1;
    }

    vector<double> calcEquation(vector<vector<string>>& equations, vector<double>& values, vector<vector<string>>& queries) {

        int n=equations.size();

        for(int i=0;i<n;i++){
            graph[equations[i][0]].push_back({equations[i][1],values[i]});
            graph[equations[i][1]].push_back({equations[i][0],1/values[i]});
            alphabet.insert(equations[i][0]);
            alphabet.insert(equations[i][1]);
        }
        vector<double> res;
        for(auto& query: queries){
            if(alphabet.find(query[0])==alphabet.end() || alphabet.find(query[1])==alphabet.end()){
                res.push_back(-1.0);
            }
            else if(query[0]==query[1]) res.push_back(1.0);
            else{
                unordered_set<string> visited;
                double ans=DFS(query[0],query[1],1,visited);
                res.push_back(ans);
            }
        }

        return res;
    }
};
```

```c
bool canPlaceFlowers(int* flowerbed, int flowerbedSize, int n) {
    int cnt=0;
    if(n==0) return true;
    if(flowerbedSize==1){
        if(flowerbed[0]==0) return true;
        else return false;
    }

    for(int i=0;i<flowerbedSize;i++){
        if(i==0 && flowerbed[0]==0 && flowerbed[1]==0){
            cnt++;
            flowerbed[i]=1;
        }
        else if(i==flowerbedSize-1 && flowerbed[flowerbedSize-1]==0 && flowerbed[flowerbedSize-2]==0){
            cnt++;
            flowerbed[i]=1;
        }
        else if(i>0 && i<flowerbedSize-1){
            if (flowerbed[i]==0 && flowerbed[i+1]==0 && flowerbed[i-1]==0){
                cnt++;
                flowerbed[i]=1;
            }
        }
    }

    return cnt>=n;
}
```
